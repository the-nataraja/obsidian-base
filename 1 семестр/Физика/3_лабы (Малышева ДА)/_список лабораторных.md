01: определение объема цилиндра

```dataviewjs
const currentFolder = dv.current().file.folder;
const pages = dv.pages(`"${currentFolder}"`)
    .where(p => p.file.name !== dv.current().file.name);

const labs = {};

for (let p of pages) {
    const match = p.file.name.match(/^(КП|ЛБ)\s*(\d{2})/i);
    
    if (match) {
        const type = match[1].toUpperCase();
        const num = match[2];
        
        if (!labs[num]) {
            labs[num] = { kp: [], lb: [] };
        }
        
        if (type === "КП") labs[num].kp.push(p.file.link);
        if (type === "ЛБ") labs[num].lb.push(p.file.link);
    }
}

const rows = Object.keys(labs).sort().map(num => [
    `**${num}**`,
    labs[num].kp.length ? labs[num].kp.join(", ") : "—",
    labs[num].lb.length ? labs[num].lb.join("<br>") : "—"
]);

dv.table(["Номер", "Конспект (КП)", "Сданные лабы (ЛБ)"], rows);
```
